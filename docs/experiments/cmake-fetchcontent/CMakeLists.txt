cmake_minimum_required(VERSION 4.4.3)
if(NOT CMAKE_VERSION VERSION_EQUAL "4.4.3")
  message(FATAL_ERROR "This experiment validates exactly CMake 4.4.3")
endif()
project(llmp_cmake_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_SKIP_RPATH ON)
set(smoke_sources "${CMAKE_CURRENT_LIST_DIR}/../toolchain-smoke")
add_executable(cpu-smoke "${smoke_sources}/main.cc")

option(SMOKE_CUDA "Build the D-032 CUDA smoke too" OFF)
if(SMOKE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 23)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(CMAKE_CUDA_EXTENSIONS OFF)
  set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
  add_executable(cuda-smoke "${smoke_sources}/main.cc" "${smoke_sources}/kernel.cu")
  target_compile_definitions(cuda-smoke PRIVATE WITH_CUDA REQUIRE_CUDA_CXX23)
  set_target_properties(cuda-smoke PROPERTIES LINKER_LANGUAGE CXX)
endif()
